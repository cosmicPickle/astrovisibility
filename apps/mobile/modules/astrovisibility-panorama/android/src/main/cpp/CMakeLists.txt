cmake_minimum_required(VERSION 3.22.1)
project(astrovisibility_panorama LANGUAGES CXX)
find_package(OpenCV REQUIRED CONFIG)
add_library(opencv_stitching STATIC IMPORTED)
set_target_properties(opencv_stitching PROPERTIES IMPORTED_LOCATION
  "${CMAKE_CURRENT_SOURCE_DIR}/../../../build/opencv-sdk/${ANDROID_ABI}/libopencv_stitching.a")
add_library(panorama_core STATIC PanoramaRegistration.cpp DirectionalPanorama.cpp)
set_target_properties(panorama_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(panorama_core PUBLIC cxx_std_17)
target_link_libraries(panorama_core PUBLIC opencv_stitching OpenCV::opencv_java4)
add_library(astrovisibility_panorama SHARED PanoramaJni.cpp)
target_link_libraries(astrovisibility_panorama PRIVATE panorama_core)
target_link_options(astrovisibility_panorama PRIVATE "-Wl,-z,max-page-size=16384")
add_executable(panorama_tests EXCLUDE_FROM_ALL PanoramaTests.cpp)
target_link_libraries(panorama_tests PRIVATE panorama_core)
